cmake_minimum_required(VERSION 3.20)
project(SWGRSuiteTests LANGUAGES CXX)
add_executable(SWGRSuiteTests SuiteTests.cpp)
target_compile_features(SWGRSuiteTests PRIVATE cxx_std_20)
target_compile_options(SWGRSuiteTests PRIVATE /W4 /WX)
target_include_directories(SWGRSuiteTests PRIVATE ../../src/mods/vr)
enable_testing()
add_executable(FF7D3D11Tests FF7D3D11Tests.cpp)
target_compile_features(FF7D3D11Tests PRIVATE cxx_std_20)
target_compile_options(FF7D3D11Tests PRIVATE /W4 /WX /utf-8)
target_include_directories(FF7D3D11Tests PRIVATE ../../src/mods/vr)
target_link_libraries(FF7D3D11Tests PRIVATE d3d11)
add_test(NAME ff7_d3d11_readback_and_recovery COMMAND FF7D3D11Tests)
add_executable(ServerPerformanceTests ServerPerformanceTests.cpp)
target_compile_features(ServerPerformanceTests PRIVATE cxx_std_20)
target_compile_options(ServerPerformanceTests PRIVATE /W4 /WX /utf-8)
target_include_directories(ServerPerformanceTests PRIVATE ../../src/mods/vr)
target_include_directories(ServerPerformanceTests PRIVATE ../../dependencies/submodules/spdlog/include)
add_test(NAME server_performance_launch_policy COMMAND ServerPerformanceTests)
add_test(NAME swgr_variant_and_mono_copy_safety COMMAND SWGRSuiteTests)
foreach(mode IN ITEMS 1 2 3 4 999)
    add_test(NAME swgr_ini_policy_${mode} COMMAND SWGRSuiteTests ${mode})
    set_tests_properties(swgr_ini_policy_${mode} PROPERTIES RESOURCE_LOCK swgr_test_ini)
endforeach()
